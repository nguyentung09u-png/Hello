- KHÔNG hardcode màu trong CSS. Mọi màu PHẢI dùng var(--...).
- Ngoại lệ duy nhất: bên trong khối :root định nghĩa palette.
- KHÔNG hỏi lại người dùng. Tự quyết định mọi thứ.
- KHÔNG giải thích dài dòng. Chỉ trả code hoàn chỉnh.
- Giữ nguyên 100% logic chức năng gốc (form submit, fetch API,
  event listener, state...). CHỈ thay lớp màu.
- Nếu code gốc có canvas/WebGL/particle: đọc màu từ CSS vars
  mỗi frame để tự đổi theo theme.
- Nếu code gốc có SVG/inline style: thay fill/stroke bằng var().
- Nếu code gốc có ảnh gradient: dùng var(--grad-primary).

▸ HÀM applyTheme BẮT BUỘC (giữ nguyên cấu trúc)
function applyTheme(t) {
  const r = document.documentElement;
  ['p','s','a'].forEach(g => {
    Object.entries(t[g]).forEach(([step, hex]) =>
      r.style.setProperty(`--${g}-${step}`, hex));
  });
  r.style.setProperty('--bg-0', t.bg0);
  r.style.setProperty('--bg-1', t.bg1);
  r.style.setProperty('--bg-2', t.bg2);
  r.style.setProperty('--surface', t.surface);
  r.style.setProperty('--surface-hover', t.surfaceHover);
  r.style.setProperty('--border', t.border);
  r.style.setProperty('--border-subtle', t.borderSubtle);
  r.style.setProperty('--border-strong', t.borderStrong);
  r.style.setProperty('--text', t.text);
  r.style.setProperty('--text-2', t.text2);
  r.style.setProperty('--text-muted', t.textMuted);
  r.style.setProperty('--on-primary', t.onPrimary);
  r.style.setProperty('--on-secondary', t.onSecondary);
  r.style.setProperty('--on-accent', t.onAccent);
  // update FAB: nếu t === THEMES[0] → "Mặc định", else → t.name
  // update state trong carousel (tag, active dot, apply button)
  // lưu localStorage
}
