Tổng quan cách làm

4 phần ghép lại:

1. HTML — 1 <button> chứa 1 <svg> inline, có aria-label cho screen reader.
2. CSS hình dáng — width/height bằng nhau + border-radius: 50% + display: flex căn giữa icon.
3. CSS nổi — position: fixed + background: linear-gradient + box-shadow 2 lớp + @keyframes pulse.
4. JS kéo — Pointer Events (pointerdown/move/up), setPointerCapture, THRESHOLD phân biệt click/drag, clamp giới hạn viewport, transform: translate() di chuyển.

Nguyên lý chính (suy luận có căn cứ):

· border-radius: 50% + width = height → hình tròn.
· position: fixed + z-index cao → luôn nổi.
· transform: translate() để di chuyển không reflow — mượt hơn left/top.
· setPointerCapture giữ sự kiện khi con trỏ ra ngoài element.
· touch-action: none để kéo trên mobile không bị cuộn trang.

Tôi không chắc:

· Kích thước 60px có tối ưu mọi ngữ cảnh — Material Design khuyến nghị 56dp.
· Ngưỡng THRESHOLD = 6px có phù hợp trên mobile — có thể cần 8–10px.

```js
(function () {
  const fab = document.getElementById('fab');
  const FAB_W = 60, FAB_H = 60;      // khớp với CSS
  const THRESHOLD = 6;                // px — phân biệt click vs drag

  let posX = 0, posY = 0;
  const clamp = (v, min, max) => v < min ? min : (v > max ? max : v);

  function render() {
    fab.style.transform = 'translate(' + posX + 'px,' + posY + 'px)';
  }
  function setPos(x, y) {
    posX = clamp(x, 0, innerWidth  - FAB_W);
    posY = clamp(y, 0, innerHeight - FAB_H);
    render();
  }

  // Vị trí ban đầu: góc dưới-phải, cách mép 20px
  setPos(innerWidth - FAB_W - 20, innerHeight - FAB_H - 20);

  // Drag state
  let dragging = false, pointerId = null;
  let startX = 0, startY = 0, baseX = 0, baseY = 0, moved = false;

  fab.addEventListener('pointerdown', (e) => {
    if (e.button !== 0 && e.pointerType === 'mouse') return;
    dragging = true; moved = false;
    pointerId = e.pointerId;
    startX = e.clientX; startY = e.clientY;
    baseX = posX; baseY = posY;
    try { fab.setPointerCapture(pointerId); } catch (err) {}
    e.preventDefault();
  });

  fab.addEventListener('pointermove', (e) => {
    if (!dragging || e.pointerId !== pointerId) return;
    const dx = e.clientX - startX;
    const dy = e.clientY - startY;
    if (!moved && (Math.abs(dx) > THRESHOLD || Math.abs(dy) > THRESHOLD)) {
      moved = true;
      fab.classList.add('dragging');
    }
    if (moved) setPos(baseX + dx, baseY + dy);
  });

  function endDrag(e) {
    if (!dragging) return;
    if (e && e.pointerId !== pointerId) return;
    dragging = false;
    fab.classList.remove('dragging');
    try { fab.releasePointerCapture(pointerId); } catch (err) {}
    pointerId = null;
  }

  fab.addEventListener('pointerup', endDrag);
  fab.addEventListener('pointercancel', endDrag);
  fab.addEventListener('lostpointercapture', endDrag);

  // Click chỉ tính khi KHÔNG kéo
  fab.addEventListener('click', (e) => {
    if (moved) { e.preventDefault(); e.stopPropagation(); return; }
    // TODO: hành động khi click
  });

  window.addEventListener('resize', () => setPos(posX, posY));
})();
```
