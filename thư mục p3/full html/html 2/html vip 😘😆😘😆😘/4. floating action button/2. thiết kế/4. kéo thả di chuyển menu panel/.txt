Tổng quan cách làm

4 phần ghép lại:

1. HTML — 1 <div class="panel"> chứa header (tay cầm kéo) + body (nội dung).
2. CSS hình dáng — position: fixed + border-radius + box-shadow + overflow: hidden.
3. CSS tay cầm — .panel-header có cursor: grab + touch-action: none + user-select: none.
4. JS kéo — Pointer Events gắn chỉ vào header (pointerdown/move/up), setPointerCapture, THRESHOLD phân biệt click/drag, clamp viewport, transform: translate().

Nguyên lý chính (suy luận có căn cứ):

· position: fixed neo vào viewport — không cuộn theo trang.
· transform: translate() di chuyển không reflow — mượt hơn left/top.
· Chỉ gắn sự kiện lên header — không đụng vào body → card/nút bên trong vẫn click được.
· setPointerCapture giữ sự kiện khi con trỏ ra ngoài header.
· clamp kẹp vị trí trong [0, W-w] × [0, H-h].
· touch-action: none trên header — kéo mobile không bị cuộn trang.

Tôi không chắc:

· Ngưỡng THRESHOLD = 6px có phù hợp trên mobile — có thể cần 8–10px.
· Bạn muốn lưu vị trí sau reload không — code dưới không lưu, có thể thêm.

```js
(function () {
  const panel  = document.getElementById('panel');
  const header = document.getElementById('panelHeader');
  const THRESHOLD = 6;

  let panelX = 0, panelY = 0;

  const clamp = (v, min, max) => v < min ? min : (v > max ? max : v);

  function setPanelPos(x, y) {
    const w = panel.offsetWidth  || 300;
    const h = panel.offsetHeight || 200;
    panelX = clamp(x, 0, window.innerWidth  - w);
    panelY = clamp(y, 0, window.innerHeight - h);
    panel.style.transform = `translate(${panelX}px, ${panelY}px)`;
  }

  // Vị trí ban đầu: góc dưới-phải, cách mép 20px
  function initPos() {
    setPanelPos(
      window.innerWidth  - panel.offsetWidth  - 20,
      window.innerHeight - panel.offsetHeight - 20
    );
  }

  // Drag state
  let dragging = false, pointerId = null, moved = false;
  let startX = 0, startY = 0, baseX = 0, baseY = 0;

  header.addEventListener('pointerdown', (e) => {
    // Bỏ qua khi click vào nút bên trong header
    if (e.target.closest('button')) return;
    if (e.button !== 0 && e.pointerType === 'mouse') return;

    dragging = true; moved = false;
    pointerId = e.pointerId;
    startX = e.clientX; startY = e.clientY;
    baseX = panelX; baseY = panelY;
    try { header.setPointerCapture(pointerId); } catch (err) {}
    e.preventDefault();
  });

  header.addEventListener('pointermove', (e) => {
    if (!dragging || e.pointerId !== pointerId) return;
    const dx = e.clientX - startX;
    const dy = e.clientY - startY;
    if (!moved && (Math.abs(dx) > THRESHOLD || Math.abs(dy) > THRESHOLD)) {
      moved = true;
      panel.classList.add('dragging');
    }
    if (moved) setPanelPos(baseX + dx, baseY + dy);
  });

  function endDrag(e) {
    if (!dragging) return;
    if (e && e.pointerId !== pointerId) return;
    dragging = false;
    panel.classList.remove('dragging');
    try { header.releasePointerCapture(pointerId); } catch (err) {}
    pointerId = null;
  }

  header.addEventListener('pointerup', endDrag);
  header.addEventListener('pointercancel', endDrag);
  header.addEventListener('lostpointercapture', endDrag);

  // Resize → clamp lại
  window.addEventListener('resize', () => setPanelPos(panelX, panelY));

  // Khởi động
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initPos);
  } else {
    initPos();
  }
  window.addEventListener('load', initPos);
})();
```

Tinh chỉnh nhanh

Muốn Sửa
Kéo từ toàn bộ panel, không chỉ header Đổi header.addEventListener → panel.addEventListener + bỏ qua button/input/a
Lưu vị trí sau reload Thêm localStorage.setItem('panel-pos', ...) trong endDrag
Ngưỡng click/drag khác THRESHOLD = 6 → 10 cho mobile
Đổi kích thước width: 300px / max-height: 60vh
Đổi vị trí ban đầu Sửa initPos()
