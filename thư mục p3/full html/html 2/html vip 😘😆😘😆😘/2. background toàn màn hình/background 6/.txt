Tổng quan cách làm

3 lớp chồng nhau:

1. Nền — body với radial-gradient lệch góc phải-trên navy → đen.
2. Glow — <div class="sky"> chứa 2 radial-gradient navy/tím + filter: blur(50px), tĩnh (không animation).
3. Canvas hạt — <canvas> chạy requestAnimationFrame, mỗi frame: spawn vệt mới (nếu đến lúc) → cập nhật vị trí → vẽ vệt → xóa vệt hết đời.

Nguyên lý vệt sao băng:

· Mỗi vệt có x, y, vx, vy, length, size, color, life, maxLife.
· Vận tốc tính từ 1 góc cố định: ANGLE = 0.78π (~140°) → tất cả cùng hướng chéo.
· Mỗi frame: x += vx, y += vy, life += 1/60.
· Fade 2 giai đoạn: 0–0.3s tăng dần, sau đó giảm dần về 0.
· Khi hết đời hoặc ra khỏi màn → splice xóa khỏi mảng.
· Spawn thêm vệt mới khi t - lastSpawn > 180 + random * 240.

Vẽ vệt = 3 lớp:

1. Đuôi: linearGradient dọc theo vệt (đậm → nhạt → trong suốt).
2. Đầu chói: chấm trắng đặc.
3. Halo đầu: radialGradient 8× size.

Tối ưu (suy luận có căn cứ):

· DPR cap 2 → nét trên retina.
· visibilitychange → dừng khi tab ẩn.
· prefers-reduced-motion → không chạy.
· Duyệt ngược khi splice để tránh bỏ sót phần tử.
· Spawn dựa vào t của requestAnimationFrame → nhịp ổn định không phụ thuộc FPS.

Tôi không chắc:

· Nhịp spawn 180–420ms có phù hợp mọi kích thước màn.
· Góc 0.78π có phải là góc tối ưu trực quan.
· splice liên tục có phân mảnh mảng khi chạy rất lâu.

```js
(function () {
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  if (!ctx) return;

  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;

  let W = 0, H = 0, dpr = 1, rafId = 0;
  let meteors = [];
  let lastSpawn = 0;

  const COLORS = ['#ffffff','#c8e0ff','#a8d0ff','#d8e8ff','#e0d8ff'];
  const ANGLE  = Math.PI * 0.78;   // ~140°, bay chéo xuống-trái

  function spawn() {
    const speed  = 4 + Math.random() * 5;
    const length = 60 + Math.random() * 140;
    const size   = 1 + Math.random() * 1.6;
    return {
      x: Math.random() * W * 1.2,
      y: -50 - Math.random() * 100,
      vx: Math.cos(ANGLE) * speed,
      vy: -Math.sin(ANGLE) * speed * -1,
      length, size,
      color: COLORS[(Math.random() * COLORS.length) | 0],
      life: 0,
      maxLife: 1.5 + Math.random() * 1.5,
    };
  }

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    c.width  = W * dpr;
    c.height = H * dpr;
    c.style.width  = W + 'px';
    c.style.height = H + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  }

  function drawMeteor(m, alpha) {
    const tailX = m.x - Math.cos(ANGLE) * m.length;
    const tailY = m.y - Math.sin(ANGLE) * m.length * -1;

    // 1. đuôi
    const g = ctx.createLinearGradient(m.x, m.y, tailX, tailY);
    g.addColorStop(0,   m.color);
    g.addColorStop(0.4, m.color + '66');
    g.addColorStop(1,   m.color + '00');

    ctx.strokeStyle = g;
    ctx.lineWidth = m.size * 1.6;
    ctx.lineCap = 'round';
    ctx.globalAlpha = alpha;
    ctx.beginPath();
    ctx.moveTo(m.x, m.y);
    ctx.lineTo(tailX, tailY);
    ctx.stroke();

    // 2. đầu chói
    ctx.globalAlpha = alpha * 0.9;
    ctx.fillStyle = '#fff';
    ctx.beginPath();
    ctx.arc(m.x, m.y, m.size * 0.8, 0, Math.PI * 2);
    ctx.fill();

    // 3. halo đầu
    const hg = ctx.createRadialGradient(m.x, m.y, 0, m.x, m.y, m.size * 8);
    hg.addColorStop(0, m.color + 'cc');
    hg.addColorStop(1, m.color + '00');
    ctx.fillStyle = hg;
    ctx.globalAlpha = alpha * 0.6;
    ctx.beginPath();
    ctx.arc(m.x, m.y, m.size * 8, 0, Math.PI * 2);
    ctx.fill();

    ctx.globalAlpha = 1;
  }

  function draw(t) {
    ctx.clearRect(0, 0, W, H);

    // spawn theo thời gian
    if (t - lastSpawn > 180 + Math.random() * 240) {
      meteors.push(spawn());
      lastSpawn = t;
    }

    // duyệt ngược để splice an toàn
    for (let i = meteors.length - 1; i >= 0; i--) {
      const m = meteors[i];
      m.x += m.vx;
      m.y += m.vy;
      m.life += 1 / 60;

      const fade = m.life < 0.3
        ? m.life / 0.3
        : Math.max(0, 1 - (m.life - 0.3) / (m.maxLife - 0.3));

      if (m.life > m.maxLife || m.x < -200 || m.y > H + 200) {
        meteors.splice(i, 1);
        continue;
      }
      drawMeteor(m, fade);
    }

    rafId = requestAnimationFrame(draw);
  }

  function start() { if (!rafId && !reduceMotion) rafId = requestAnimationFrame(draw); }
  function stop()  { if (rafId) { cancelAnimationFrame(rafId); rafId = 0; } }

  window.addEventListener('resize', resize, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) stop();
    else start();
  });

  resize();
  start();
})();
```
