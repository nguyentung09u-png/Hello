Tổng quan cách làm

3 lớp chồng nhau:

1. Nền — body với linear-gradient dọc navy → xanh biển → xanh đen.
2. Glow — <div class="caustic"> chứa 2 radial-gradient cyan/xanh + filter: blur(50px) + animation drift.
3. Canvas hạt — <canvas> chạy requestAnimationFrame, mỗi frame: cập nhật vị trí → tái sinh nếu ra khỏi đỉnh → vẽ 3 lớp.

Nguyên lý bong bóng:

· Mỗi bong có x, y, r, vx, vy, sway, swaySpeed, alpha, hue.
· vy phụ thuộc bán kính: -(0.6 + random*1.2) - r * 0.03 → bong lớn bay nhanh hơn.
· Mỗi frame: sway += 0.03 * swaySpeed, x += vx + sin(sway)*0.4, y += vy.
· Ra khỏi đỉnh → tái sinh ở đáy với bán kính mới.

Vẽ 1 bong = 3 lớp:

1. Thân — createRadialGradient ngược: trong suốt ở tâm → đậm ở rìa (mô phỏng màng mỏng).
2. Viền mỏng — stroke alpha cao, lineWidth 0.8.
3. Highlight — createRadialGradient trắng ở góc trên-trái (−0.35r, −0.35r), bán kính 0.5r.

Tối ưu (suy luận có căn cứ):

· DPR cap 2.
· visibilitychange → dừng khi tab ẩn.
· prefers-reduced-motion → không chạy.
· Số bong cap 50 (vì mỗi bong vẽ 3 gradient).

Tôi không chắc:

· Hệ số - r * 0.03 cho vận tốc là tối ưu trực quan.
· Vị trí highlight (−0.35r, −0.35r) với bán kính 0.5r cho cảm giác "tròn" hay không.

```js
(function () {
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  if (!ctx) return;

  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let W = 0, H = 0, dpr = 1, rafId = 0;
  let bubbles = [];

  function count() { return Math.min(50, Math.max(18, Math.floor(W / 30))); }

  function make(anywhere) {
    const r = 4 + Math.random() * 14;
    return {
      x: Math.random() * W,
      y: anywhere ? Math.random() * H : H + r,
      r,
      vy: -(0.6 + Math.random() * 1.2) - r * 0.03,   // bong lớn bay nhanh hơn
      vx: (Math.random() - 0.5) * 0.3,
      sway: Math.random() * Math.PI * 2,
      swaySpeed: 0.4 + Math.random() * 0.7,
      alpha: 0.35 + Math.random() * 0.35,
      hue: 190 + Math.random() * 40,                 // 190 → 230
    };
  }

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    c.width  = W * dpr;
    c.height = H * dpr;
    c.style.width  = W + 'px';
    c.style.height = H + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const need = count();
    while (bubbles.length < need) bubbles.push(make(true));
    if (bubbles.length > need) bubbles.length = need;
  }

  function drawBubble(b) {
    const hue   = b.hue;
    const alpha = b.alpha;

    // 1. thân — radial gradient ngược (trong suốt ở tâm, đậm ở rìa)
    const g = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r);
    g.addColorStop(0,   'hsla(' + hue + ', 90%, 70%, 0)');
    g.addColorStop(0.7, 'hsla(' + hue + ', 90%, 70%, ' + (alpha * 0.3) + ')');
    g.addColorStop(1,   'hsla(' + hue + ', 90%, 80%, ' + alpha + ')');
    ctx.fillStyle = g;
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
    ctx.fill();

    // 2. viền mỏng
    ctx.strokeStyle = 'hsla(' + hue + ', 90%, 85%, ' + (alpha * 0.9) + ')';
    ctx.lineWidth = 0.8;
    ctx.beginPath();
    ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
    ctx.stroke();

    // 3. highlight góc trên-trái
    const hx = b.x - b.r * 0.35;
    const hy = b.y - b.r * 0.35;
    const hg = ctx.createRadialGradient(hx, hy, 0, hx, hy, b.r * 0.5);
    hg.addColorStop(0, 'rgba(255,255,255,' + (alpha * 1.3) + ')');
    hg.addColorStop(1, 'rgba(255,255,255,0)');
    ctx.fillStyle = hg;
    ctx.beginPath();
    ctx.arc(hx, hy, b.r * 0.5, 0, Math.PI * 2);
    ctx.fill();
  }

  function draw(t) {
    ctx.clearRect(0, 0, W, H);

    for (let i = 0; i < bubbles.length; i++) {
      const b = bubbles[i];
      b.sway += 0.03 * b.swaySpeed;
      b.x += b.vx + Math.sin(b.sway) * 0.4;
      b.y += b.vy;

      if (b.y + b.r < -10) {
        bubbles[i] = make(false);
        continue;
      }

      drawBubble(b);
    }

    rafId = requestAnimationFrame(draw);
  }

  function start() { if (!rafId && !reduceMotion) rafId = requestAnimationFrame(draw); }
  function stop()  { if (rafId) { cancelAnimationFrame(rafId); rafId = 0; } }

  window.addEventListener('resize', resize, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) stop();
    else start();
  });

  resize();
  start();
})();
```
