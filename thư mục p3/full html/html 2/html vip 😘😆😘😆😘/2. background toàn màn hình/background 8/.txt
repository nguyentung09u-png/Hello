Tổng quan cách làm

3 lớp chồng nhau:

1. Nền — body với radial-gradient tỏa tròn từ tâm, navy → đen.
2. Glow — <div class="glow"> chứa 2 radial-gradient navy/tím + filter: blur(60px), tĩnh.
3. Canvas hạt — <canvas> chạy requestAnimationFrame, mỗi frame: cập nhật vị trí → dựng spatial hash → vẽ đường nối → vẽ node.

Nguyên lý node:

· Mỗi node có x, y, vx, vy, r, phase, speed.
· Mỗi frame: x += vx, y += vy, wrap-around khi ra mép.
· Nhấp nháy: a = 0.55 + 0.45 * Math.sin(t * speed + phase) — lệch pha mỗi node.
· Halo createRadialGradient bán kính 6× node.

Spatial hash grid (nâng cấp chính):

· Chia không gian thành ô vuông cạnh CELL = MAX_D.
· Mỗi node rơi vào 1 ô theo (x/CELL|0, y/CELL|0).
· Khi tìm hàng xóm, chỉ so với node trong 9 ô xung quanh (3×3).
· Từ O(n²) → ~O(n).

Đường nối:

· Chỉ vẽ nếu khoảng cách < MAX_D (150).
· Alpha giảm dần: 0.28 * (1 - d / MAX_D).
· Hue đổi theo hướng ngang: hue = 200 + (dx / MAX_D) * 60 → dải 140 → 260.

Tối ưu (suy luận có căn cứ):

· DPR cap 2.
· visibilitychange → dừng khi tab ẩn.
· prefers-reduced-motion → không chạy.
· So sánh key chuỗi để tránh vẽ trùng cặp.

Tôi không chắc:

· Logic nkey < key (so sánh chuỗi) có xử lý đúng 100% mọi cấu hình — có thể bỏ sót hoặc trùng ở vài case.
· Spatial hash có nhanh hơn O(n²) ở 140 node hay không — lợi thế rõ khi node 300+.

```js
(function () {
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  if (!ctx) return;

  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;
  let W = 0, H = 0, dpr = 1, rafId = 0;
  let nodes = [];

  const MAX_D = 150;
  const CELL  = MAX_D;

  function count() {
    return Math.min(140, Math.max(50, Math.floor(W * H / 18000)));
  }

  function make() {
    return {
      x: Math.random() * W,
      y: Math.random() * H,
      vx: (Math.random() - 0.5) * 0.4,
      vy: (Math.random() - 0.5) * 0.4,
      r: 1 + Math.random() * 1.4,
      phase: Math.random() * Math.PI * 2,
      speed: 0.8 + Math.random() * 1.4,
    };
  }

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    c.width  = W * dpr;
    c.height = H * dpr;
    c.style.width  = W + 'px';
    c.style.height = H + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const need = count();
    while (nodes.length < need) nodes.push(make());
    if (nodes.length > need) nodes.length = need;
  }

  // Spatial hash: Map "cx,cy" → mảng node trong ô
  function buildGrid() {
    const grid = new Map();
    for (const n of nodes) {
      const cx = (n.x / CELL) | 0;
      const cy = (n.y / CELL) | 0;
      const key = cx + ',' + cy;
      let arr = grid.get(key);
      if (!arr) { arr = []; grid.set(key, arr); }
      arr.push(n);
    }
    return grid;
  }

  function drawConnections(grid) {
    for (const [key, list] of grid) {
      const parts = key.split(',');
      const cx = +parts[0], cy = +parts[1];

      for (let ox = -1; ox <= 1; ox++) {
        for (let oy = -1; oy <= 1; oy++) {
          const nkey = (cx + ox) + ',' + (cy + oy);
          const other = grid.get(nkey);
          if (!other) continue;

          for (let i = 0; i < list.length; i++) {
            const a = list[i];
            for (let j = 0; j < other.length; j++) {
              const b = other[j];
              if (a === b) continue;
              if (nkey < key) continue;
              if (nkey === key && j <= i) continue;

              const dx = a.x - b.x, dy = a.y - b.y;
              const d2 = dx*dx + dy*dy;
              if (d2 < MAX_D * MAX_D) {
                const d = Math.sqrt(d2);
                const alpha = 0.28 * (1 - d / MAX_D);
                const hue = 200 + (dx / MAX_D) * 60;
                ctx.strokeStyle = 'hsla(' + hue + ', 90%, 70%, ' + alpha.toFixed(3) + ')';
                ctx.lineWidth = 0.7;
                ctx.beginPath();
                ctx.moveTo(a.x, a.y);
                ctx.lineTo(b.x, b.y);
                ctx.stroke();
              }
            }
          }
        }
      }
    }
  }

  function draw(t) {
    ctx.clearRect(0, 0, W, H);

    // cập nhật vị trí
    for (const n of nodes) {
      n.x += n.vx; n.y += n.vy;
      if (n.x < -10) n.x = W + 10;
      else if (n.x > W + 10) n.x = -10;
      if (n.y < -10) n.y = H + 10;
      else if (n.y > H + 10) n.y = -10;
    }

    // đường nối
    const grid = buildGrid();
    drawConnections(grid);

    // node
    for (const n of nodes) {
      const a = 0.55 + 0.45 * Math.sin(t * 0.001 * n.speed + n.phase);

      const g = ctx.createRadialGradient(n.x, n.y, 0, n.x, n.y, n.r * 6);
      g.addColorStop(0, 'rgba(160,200,255,0.9)');
      g.addColorStop(1, 'rgba(160,200,255,0)');
      ctx.fillStyle = g;
      ctx.beginPath();
      ctx.arc(n.x, n.y, n.r * 6, 0, Math.PI * 2);
      ctx.fill();

      ctx.beginPath();
      ctx.fillStyle = '#d8e8ff';
      ctx.globalAlpha = a;
      ctx.arc(n.x, n.y, n.r, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.globalAlpha = 1;

    rafId = requestAnimationFrame(draw);
  }

  function start() { if (!rafId && !reduceMotion) rafId = requestAnimationFrame(draw); }
  function stop()  { if (rafId) { cancelAnimationFrame(rafId); rafId = 0; } }

  window.addEventListener('resize', resize, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) stop();
    else start();
  });

  resize();
  start();
})();
```

Lưu ý (phân biệt rõ):

· Suy luận có căn cứ: spatial hash là kỹ thuật chuẩn để tìm hàng xóm gần trong không gian 2D; Canvas 2D + requestAnimationFrame chạy trên trình duyệt hiện đại.
· Tôi không chắc so sánh key chuỗi nkey < key xử lý đúng 100% mọi cấu hình — nên thay bằng so sánh id số nếu cần chắc chắn.
· Tôi không chắc lợi thế của spatial hash ở 140 node có rõ — cần đo thực tế.

🎯 Tóm 1 câu: Nền radial-gradient navy + glow tĩnh + canvas node dùng spatial hash grid (~O(n)) để nối đường, đường đổi hue theo hướng ngang, mỗi node có halo 6× và nhấp nháy lệch pha — chạy requestAnimationFrame với DPR cap, visibility, reduced-motion.
