Tổng quan cách làm

Dựng 2 lớp phủ toàn màn hình chồng lên nền đen:

· Lớp dưới — .blur-background chứa 3 .blob: position: fixed, filter: blur(100px), mix-blend-mode: screen, animation floatBlob (dịch + scale + xoay) và colorShift (đổi hue-rotate + độ blur).
· Lớp trên — <canvas> vẽ hạt bằng JS: requestAnimationFrame, cập nhật vị trí mỗi frame, vẽ đường nối khi hạt gần nhau, halo gradient cho hạt lớn, DPR cho nét, tạm dừng khi tab ẩn.

Nguyên lý:

· position: fixed + pointer-events: none để không chiếm layout, không chặn click.
· z-index âm cho blob, 0 cho canvas, 1+ cho nội dung.
· Mỗi frame: cộng vận tốc vào vị trí hạt → vẽ lại toàn bộ canvas.

Lưu ý (phân biệt rõ):

· Suy luận có căn cứ: position: fixed tránh scrollbar; z-index âm giữ blob dưới nội dung.
· Tôi không chắc mix-blend-mode: screen + filter: blur(100px) hiển thị giống nhau 100% trên Safari cũ.
· Tôi không chắc mask-composite không dùng ở đây nên không ảnh hưởng.

```js
(function () {
  const canvas = document.getElementById('particle-canvas');
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  if (!ctx) return;

  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;

  let W = 0, H = 0, dpr = 1, rafId = 0;
  let particles = [];

  const COLORS = ['#4a8ec2','#2a6ba8','#7ab8e0','#1a4a8a','#5aa0d8',
                  '#3f7eb6','#9fd0f0','#0f3a6e','#6bb0e8','#2c5f9a'];

  function count() {
    return Math.min(120, Math.max(50, Math.floor((W * H) / 14000)));
  }
  function make(randomY) {
    const x = Math.random() * W;
    const y = randomY
      ? Math.random() * H
      : (Math.random() < 0.3 ? -10 : H + 10);
    const baseSize = Math.random() * 3.2 + 0.8;
    return {
      x, y,
      vx: (Math.random() - 0.5) * 0.6,
      vy: (Math.random() * 0.8 + 0.2) * (Math.random() < 0.5 ? -1 : 1),
      baseSize,
      pulsePhase: Math.random() * Math.PI * 2,
      pulseSpeed: 0.02 + Math.random() * 0.05,
      color: COLORS[(Math.random() * COLORS.length) | 0],
      glow: baseSize > 2.8,
      alpha: 0.6 + Math.random() * 0.4,
    };
  }

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    canvas.width = W * dpr;
    canvas.height = H * dpr;
    canvas.style.width = W + 'px';
    canvas.style.height = H + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const need = count();
    while (particles.length < need) particles.push(make(true));
    if (particles.length > need) particles.length = need;
  }

  function connections(maxD) {
    for (let i = 0; i < particles.length; i++) {
      const a = particles[i];
      for (let j = i + 1; j < particles.length; j++) {
        const b = particles[j];
        const dx = a.x - b.x, dy = a.y - b.y;
        const d2 = dx*dx + dy*dy;
        if (d2 < maxD * maxD) {
          const alpha = 0.18 * (1 - Math.sqrt(d2) / maxD);
          ctx.strokeStyle = 'rgba(80,150,220,' + alpha.toFixed(3) + ')';
          ctx.lineWidth = 0.7;
          ctx.beginPath();
          ctx.moveTo(a.x, a.y);
          ctx.lineTo(b.x, b.y);
          ctx.stroke();
        }
      }
    }
  }

  function draw(t) {
    ctx.clearRect(0, 0, W, H);
    connections(120);

    for (const p of particles) {
      const pulse = 0.8 + 0.4 * Math.sin(p.pulsePhase + t * 0.001 * p.pulseSpeed);
      const size = p.baseSize * pulse;
      p.pulsePhase += 0.03;
      if (p.pulsePhase > Math.PI * 200) p.pulsePhase -= Math.PI * 200;

      if (p.glow) {
        const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, size * 4);
        g.addColorStop(0, p.color + 'aa');
        g.addColorStop(0.6, p.color + '55');
        g.addColorStop(1, p.color + '00');
        ctx.beginPath();
        ctx.fillStyle = g;
        ctx.arc(p.x, p.y, size * 4, 0, Math.PI * 2);
        ctx.fill();
      }

      ctx.beginPath();
      ctx.fillStyle = p.color;
      ctx.globalAlpha = p.alpha * (0.9 + 0.2 * Math.sin(p.pulsePhase));
      ctx.arc(p.x, p.y, size, 0, Math.PI * 2);
      ctx.fill();
      ctx.globalAlpha = 1;
    }
  }

  function update() {
    for (const p of particles) {
      p.x += p.vx; p.y += p.vy;
      if (p.x < -20) p.x = W + 15;
      else if (p.x > W + 20) p.x = -15;
      if (p.y < -20) p.y = H + 15;
      else if (p.y > H + 20) p.y = -15;
    }
  }

  function animate(t) {
    update(); draw(t);
    rafId = requestAnimationFrame(animate);
  }
  function start() { if (!rafId && !reduceMotion) rafId = requestAnimationFrame(animate); }
  function stop()  { if (rafId) { cancelAnimationFrame(rafId); rafId = 0; } }

  window.addEventListener('resize', resize, { passive: true });
  window.addEventListener('orientationchange', resize, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) stop(); else start();
  });

  resize();
  start();
})();
```

Lưu ý (phân biệt rõ):

· Suy luận có căn cứ: mỗi frame gọi clearRect rồi vẽ lại toàn bộ — không lưu vệt.
· Tôi không chắc hiệu năng drawConnections khi số hạt lên 200+ vì là O(n²).
· Tôi không chắc mix-blend-mode: screen hoạt động giống nhau trên Safari cũ.

🎯 Tóm 1 câu: 2 lớp position: fixed — blob navy (blur 100px + hue-rotate + floatBlob) ở z-index: -3, canvas hạt (requestAnimationFrame + nối đường + halo + DPR + visibility + reduced-motion) ở z-index: 0, nội dung ở z-index: 1.
