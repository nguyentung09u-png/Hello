Tổng quan cách làm

3 lớp chồng nhau:

1. Nền — body với radial-gradient lệch tâm navy → đen.
2. Glow — <div class="glow-bg"> chứa 2 radial-gradient + filter: blur(50px) + animation drift.
3. Canvas hạt — <canvas> chạy requestAnimationFrame, mỗi frame: cập nhật vị trí → vẽ đường nối → vẽ hạt.

Nguyên lý hạt:

· Mỗi hạt có x, y, vx, vy, r, color, phase, speed.
· Mỗi frame: x += vx; y += vy.
· Ra khỏi mép → wrap-around: xuất hiện ở mép đối diện (không tái sinh).
· Mọi hạt đều có halo bằng createRadialGradient (5× bán kính).
· Nhấp nháy: alpha = 0.55 + 0.45 * Math.sin(t * speed + phase) — mỗi hạt lệch pha.

Đường nối:

· Duyệt cặp (i, j) — O(n²).
· Nếu khoảng cách < 130px → vẽ đường.
· Alpha giảm dần theo khoảng cách: 0.14 * (1 - d / maxD).
· Màu cố định rgba(140,180,255,...).

Tối ưu (suy luận có căn cứ):

· Cap số hạt 110 để hạn chế O(n²).
· DPR cap 2 → nét trên retina mà không quá nặng.
· visibilitychange → dừng khi tab ẩn.
· prefers-reduced-motion → không chạy.

Tôi không chắc:

· Alpha 0.14 có phải giá trị tối ưu cho đường nối hay không.
· Hiệu năng khi số hạt vượt 150 — chưa đo thực tế.

```js
(function () {
  const c = document.getElementById('c');
  const ctx = c.getContext('2d');
  if (!ctx) return;

  const reduceMotion = matchMedia('(prefers-reduced-motion: reduce)').matches;

  let W = 0, H = 0, dpr = 1, rafId = 0;
  let parts = [];

  const COLORS = ['#7ab8ff','#a0d4ff','#c8a0ff','#9fbfff','#d8b8ff','#a8e0ff'];
  const MAX_D = 130;

  function count() {
    return Math.min(110, Math.max(50, Math.floor(W * H / 22000)));
  }

  function make() {
    return {
      x: Math.random() * W,
      y: Math.random() * H,
      vx: (Math.random() - 0.5) * 0.25,
      vy: (Math.random() - 0.5) * 0.25,
      r: Math.random() * 1.3 + 0.5,
      color: COLORS[(Math.random() * COLORS.length) | 0],
      phase: Math.random() * Math.PI * 2,
      speed: 0.5 + Math.random() * 1.2,
    };
  }

  function resize() {
    dpr = Math.min(window.devicePixelRatio || 1, 2);
    W = window.innerWidth;
    H = window.innerHeight;
    c.width  = W * dpr;
    c.height = H * dpr;
    c.style.width  = W + 'px';
    c.style.height = H + 'px';
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);

    const need = count();
    while (parts.length < need) parts.push(make());
    if (parts.length > need) parts.length = need;
  }

  function connections() {
    for (let i = 0; i < parts.length; i++) {
      const a = parts[i];
      for (let j = i + 1; j < parts.length; j++) {
        const b = parts[j];
        const dx = a.x - b.x, dy = a.y - b.y;
        const d2 = dx*dx + dy*dy;
        if (d2 < MAX_D * MAX_D) {
          const alpha = 0.14 * (1 - Math.sqrt(d2) / MAX_D);
          ctx.strokeStyle = 'rgba(140,180,255,' + alpha.toFixed(3) + ')';
          ctx.lineWidth = 0.6;
          ctx.beginPath();
          ctx.moveTo(a.x, a.y);
          ctx.lineTo(b.x, b.y);
          ctx.stroke();
        }
      }
    }
  }

  function draw(t) {
    ctx.clearRect(0, 0, W, H);

    // cập nhật vị trí
    for (const p of parts) {
      p.x += p.vx;
      p.y += p.vy;
      if (p.x < -10) p.x = W + 10;
      else if (p.x > W + 10) p.x = -10;
      if (p.y < -10) p.y = H + 10;
      else if (p.y > H + 10) p.y = -10;
    }

    // đường nối
    connections();

    // hạt
    for (const p of parts) {
      const a = 0.55 + 0.45 * Math.sin(t * 0.001 * p.speed + p.phase);

      // halo
      const g = ctx.createRadialGradient(p.x, p.y, 0, p.x, p.y, p.r * 5);
      g.addColorStop(0,   p.color + '99');
      g.addColorStop(0.5, p.color + '22');
      g.addColorStop(1,   p.color + '00');
      ctx.fillStyle = g;
      ctx.beginPath();
      ctx.arc(p.x, p.y, p.r * 5, 0, Math.PI * 2);
      ctx.fill();

      // lõi
      ctx.beginPath();
      ctx.fillStyle = p.color;
      ctx.globalAlpha = a;
      ctx.arc(p.x, p.y, p.r, 0, Math.PI * 2);
      ctx.fill();
    }
    ctx.globalAlpha = 1;

    rafId = requestAnimationFrame(draw);
  }

  function start() { if (!rafId && !reduceMotion) rafId = requestAnimationFrame(draw); }
  function stop()  { if (rafId) { cancelAnimationFrame(rafId); rafId = 0; } }

  window.addEventListener('resize', resize, { passive: true });
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) stop();
    else start();
  });

  resize();
  start();
})();
```

Lưu ý (phân biệt rõ):

· Suy luận có căn cứ: dùng Canvas 2D + requestAnimationFrame tiêu chuẩn, chạy được trên Chrome/Edge/Firefox/Safari hiện đại.
· Tôi không chắc alpha 0.14 là tối ưu — có thể cần chỉnh theo gu thẩm mỹ.
· Tôi không chắc drawConnections O(n²) với 110 hạt có ổn định trên máy yếu — đã giới hạn số hạt để giảm tải.
