Tổng quan cách làm

Crosshair

1. Lưu state: một object aim = { x, y, active }.
2. Cập nhật state: khi pointerdown/pointermove → đổi aim.x, aim.y; pointerup/pointercancel → active = false.
3. Vẽ mỗi frame:
   · ctx.save()
   · ctx.translate(aim.x, aim.y) — dời gốc về tâm ngắm.
   · ctx.rotate(time) — xoay cả vòng (nếu muốn).
   · Vẽ 4 vạch: dùng vòng lặp, mỗi lần vẽ 1 vạch rồi ctx.rotate(π/2).
   · Vẽ thêm 1 arc nhỏ ở tâm.
   · ctx.restore()

Projectile (mũi tên)

1. State: mảng projectiles = [], mỗi phần tử { x, y, vx, vy, life }.
2. Sinh ra: từ vị trí bắn (sx, sy) và đích (tx, ty):
   · dx = tx - sx, dy = ty - sy
   · len = Math.hypot(dx, dy) || 1
   · vx = dx/len * speed, vy = dy/len * speed
3. Cập nhật mỗi frame:
   · x += vx * dt, y += vy * dt
   · life -= dt
   · Lọc bỏ khi life <= 0 hoặc ra ngoài màn hình.
4. Vẽ mỗi frame:
   · ctx.save()
   · ctx.translate(p.x, p.y)
   · ctx.rotate(Math.atan2(p.vy, p.vx)) — xoay theo hướng bay.
   · Vẽ thân + đầu + đuôi bằng path.
   · ctx.restore()

---

Code chung chung

```js
// ===== STATE =====
const W = 450, H = 800;
const aim = { x: 225, y: 300, active: false };
const projectiles = [];
const SPEED = 700;

// ===== TẠO PROJECTILE =====
function fire(sx, sy, tx, ty) {
  const dx = tx - sx, dy = ty - sy;
  const len = Math.hypot(dx, dy) || 1;
  projectiles.push({
    x: sx, y: sy,
    vx: dx / len * SPEED,
    vy: dy / len * SPEED,
    life: 2.0
  });
}

// ===== UPDATE =====
function updateProjectiles(dt) {
  for (const p of projectiles) {
    p.x += p.vx * dt;
    p.y += p.vy * dt;
    p.life -= dt;
  }
  for (let i = projectiles.length - 1; i >= 0; i--) {
    const p = projectiles[i];
    if (p.life <= 0 || p.x < -60 || p.x > W + 60 || p.y < -60 || p.y > H + 60) {
      projectiles.splice(i, 1);
    }
  }
}

// ===== VẼ CROSSHAIR =====
function drawCrosshair(t) {
  if (!aim.active) return;
  const pulse = 0.5 + Math.sin(t * 10) * 0.5;

  ctx.save();
  ctx.translate(aim.x, aim.y);
  ctx.rotate(t * 1.5);
  ctx.strokeStyle = `rgba(255,215,110,${0.5 + pulse * 0.4})`;
  ctx.lineWidth = 1.6;
  ctx.shadowColor = 'rgba(255,215,110,.9)';
  ctx.shadowBlur = 10;

  const r = 13;
  for (let i = 0; i < 4; i++) {
    ctx.beginPath();
    ctx.moveTo(r, 0);
    ctx.lineTo(r * 0.55, 0);
    ctx.stroke();
    ctx.rotate(Math.PI / 2);
  }
  ctx.beginPath();
  ctx.arc(0, 0, 3, 0, Math.PI * 2);
  ctx.stroke();

  ctx.restore();
}

// ===== VẼ PROJECTILE =====
function drawProjectiles() {
  for (const p of projectiles) {
    const ang = Math.atan2(p.vy, p.vx);
    ctx.save();
    ctx.translate(p.x, p.y);
    ctx.rotate(ang);

    ctx.shadowColor = '#ffd166';
    ctx.shadowBlur = 12;
    ctx.fillStyle = '#ffd166';

    // thân
    ctx.fillRect(-11, -1.4, 22, 2.8);

    // đầu nhọn
    ctx.beginPath();
    ctx.moveTo(13, 0);
    ctx.lineTo(6, -4);
    ctx.lineTo(6, 4);
    ctx.closePath();
    ctx.fill();

    // đuôi
    ctx.beginPath();
    ctx.moveTo(-11, 0);
    ctx.lineTo(-15, -4);
    ctx.lineTo(-8, -1.4);
    ctx.closePath();
    ctx.fill();

    ctx.restore();
  }
}

// ===== INPUT =====
cvs.addEventListener('pointerdown', e => {
  const r = cvs.getBoundingClientRect();
  aim.x = (e.clientX - r.left) / r.width * W;
  aim.y = (e.clientY - r.top) / r.height * H;
  aim.active = true;
  fire(225, 608, aim.x, aim.y);
});
cvs.addEventListener('pointermove', e => {
  if (!aim.active) return;
  const r = cvs.getBoundingClientRect();
  aim.x = (e.clientX - r.left) / r.width * W;
  aim.y = (e.clientY - r.top) / r.height * H;
});
cvs.addEventListener('pointerup', () => aim.active = false);
cvs.addEventListener('pointercancel', () => aim.active = false);

// ===== VÒNG LẶP =====
let last = 0;
function loop(ts) {
  const t = ts / 1000;
  const dt = Math.min((ts - last) / 1000, 0.05);
  last = ts;

  updateProjectiles(dt);
  // ... vẽ nền ở đây ...
  drawProjectiles();
  drawCrosshair(t);

  requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
```

---

🎯 Tóm 1 câu: Crosshair = object {x,y,active} vẽ bằng translate + rotate(time) rồi lặp rotate(π/2) 4 lần cho 4 vạch; Projectile = mảng {x,y,vx,vy,life} cập nhật x += vx*dt, vẽ bằng translate + rotate(atan2(vy,vx)) để mũi tên luôn xoay đúng hướng bay.
